void Log(const std::string& line) { lifecycle::Log("preyvr_weapon_sound " + line); }

// Every function hooked through a named RVA, and the slots that lead to them, are
// ones the gate checks.
static_assert(contract::Gated(sn::Contracts, sn::PlaySoundEnter) && contract::Gated(sn::Contracts, sn::GetAudioTriggerId) &&
              contract::Gated(sn::Contracts, sn::TriggerIdByName) && contract::Gated(sn::Contracts, sn::ProxyExecuteTrigger));

constexpr std::uint64_t kWindowNs = 3000000000ull;
constexpr unsigned kMaxLines = 200;

std::atomic<bool> wanted{false}, installed{false}, hookFailed{false}, fault{false};
std::atomic<const char*> gate{"unchecked"}, audioSystem{"unread"};
// The open window: its start (0 when none) and end, and the weapon it is for.
std::atomic<std::uint64_t> windowStart{0}, windowEnd{0};
std::atomic<std::uint32_t> windowItem{0};
std::atomic<unsigned> windowLines{0};
std::atomic<std::uint64_t> windows{0}, clips{0}, resolves{0}, executes{0}, dropped{0};

using ClipEnterFn = void(__fastcall*)(std::uintptr_t, float, float, std::uintptr_t);
using TriggerIdFn = bool(__fastcall*)(std::uintptr_t, const char*, std::uint32_t*);
using ExecuteFn = bool(__fastcall*)(std::uintptr_t, std::uint32_t, int, std::uint32_t, const void*);
using EntityIdFn = std::uint32_t(__fastcall*)(std::uintptr_t);
std::atomic<ClipEnterFn> originalClipEnter{nullptr};
std::atomic<TriggerIdFn> originalTriggerId{nullptr};
std::atomic<ExecuteFn> originalExecute{nullptr};

// --- reads, guarded (engine memory, from any thread) -----------------------------
// A C string, at most 95 characters: "" for none, "?" when unreadable.
void CopyName(const char* text, char (&out)[96])
{
    __try {
        std::size_t i = 0;
        for (; text && i + 1 < sizeof(out) && text[i]; ++i) {
            out[i] = text[i] > ' ' && text[i] < 127 ? text[i] : '_';
        }
        out[i] = 0;
    } __except (EXCEPTION_EXECUTE_HANDLER) {
        out[0] = '?';
        out[1] = 0;
    }
}
// An IEntity's id (its vtable +0x08), or 0.
std::uint32_t EntityId(std::uintptr_t entity)
{
    __try {
        if (!entity) { return 0; }
        const auto table = *reinterpret_cast<const std::uintptr_t* const*>(entity);
        return reinterpret_cast<EntityIdFn>(table[sn::EntityIdSlot / sizeof(void*)])(entity);
    } __except (EXCEPTION_EXECUTE_HANDLER) {
        return 0;
    }
}
template <class T> T ReadOr(std::uintptr_t at, T fallback)
{
    __try {
        return at ? *reinterpret_cast<const T*>(at) : fallback;
    } __except (EXCEPTION_EXECUTE_HANDLER) {
        return fallback;
    }
}
struct Clip {
    std::uintptr_t entity = 0;
    char start[96]{}, stop[96]{};
    std::uint32_t joint = 0, flags = 0;
    bool fireAndForget = false;
};
void ReadClip(std::uintptr_t clip, std::uintptr_t params, Clip& c)
{
    c.entity = ReadOr<std::uintptr_t>(clip + sn::ClipEntity, 0);
    CopyName(ReadOr<const char*>(params + sn::StartTrigger, nullptr), c.start);
    CopyName(ReadOr<const char*>(params + sn::StopTrigger, nullptr), c.stop);
    c.joint = ReadOr<std::uint32_t>(params + sn::AttachmentJoint, 0);
    c.flags = ReadOr<std::uint32_t>(params + sn::SoundFlags, 0);
    c.fireAndForget = ReadOr<std::uint8_t>(params + sn::FireAndForget, 0) != 0;
}

// --- the window -------------------------------------------------------------------
// Milliseconds into the open window, counting this line; -1 when none is open or
// the window has had its lines.
long long WindowMs()
{
    const auto start = windowStart.load();
    if (!start) { return -1; }
    const auto now = MonotonicNanoseconds();
    if (now > windowEnd.load()) { return -1; }
    if (windowLines.fetch_add(1) >= kMaxLines) {
        ++dropped;
        return -1;
    }
    return static_cast<long long>((now - start) / 1000000ull);
}
std::string Who(std::uint32_t id)
{
    if (id == sn::PlayerEntityId) { return "player"; }
    if (id && id == windowItem.load()) { return "weapon"; }
    return std::to_string(id);
}
std::string Ms(long long ms) { return std::to_string(ms); }

// --- the hooks --------------------------------------------------------------------
void __fastcall ClipEnterHook(std::uintptr_t clip, float blend, float duration, std::uintptr_t params)
{
    if (const auto ms = WindowMs(); ms >= 0) {
        Clip c{};
        ReadClip(clip, params, c);
        ++clips;
        Log("clip tMs=" + Ms(ms) + " entity=" + Who(EntityId(c.entity)) + " start=" + (c.start[0] ? c.start : "-") +
            " stop=" + (c.stop[0] ? c.stop : "-") + " fireAndForget=" + std::to_string(c.fireAndForget) +
            " soundFlags=" + std::to_string(c.flags) + " joint=" + Hex(c.joint) + " blendMs=" +
            std::to_string(static_cast<int>(blend * 1000)) + " durationMs=" + std::to_string(static_cast<int>(duration * 1000)));
    }
    originalClipEnter.load()(clip, blend, duration, params);
}
bool __fastcall TriggerIdHook(std::uintptr_t atl, const char* name, std::uint32_t* id)
{
    const bool found = originalTriggerId.load()(atl, name, id);
    if (const auto ms = WindowMs(); ms >= 0) {
        char text[96]{};
        CopyName(name, text);
        ++resolves;
        Log("resolve tMs=" + Ms(ms) + " name=" + (text[0] ? text : "-") + " id=" +
            Hex(ReadOr<std::uint32_t>(reinterpret_cast<std::uintptr_t>(id), 0)) + " found=" + std::to_string(found));
    }
    return found;
}
bool __fastcall ExecuteHook(std::uintptr_t proxy, std::uint32_t id, int lipSync, std::uint32_t local, const void* callbacks)
{
    if (const auto ms = WindowMs(); ms >= 0) {
        ++executes;
        Log("execute tMs=" + Ms(ms) + " id=" + Hex(id) + " entity=" +
            Who(EntityId(ReadOr<std::uintptr_t>(proxy + sn::ProxyEntity, 0))) + " proxy=" + std::to_string(local) +
            " lipSync=" + std::to_string(lipSync));
    }
    return originalExecute.load()(proxy, id, lipSync, local, callbacks);
}

// --- the gate (main thread), shared by the capture and the replay ------------------
std::uintptr_t SlotTarget(std::uintptr_t base, std::uintptr_t table, std::size_t slot)
{
    return *reinterpret_cast<const std::uintptr_t*>(base + table + slot) - base;
}
bool GateBodies(std::uintptr_t base)
{
    __try {
        return contract::BodiesMatch(base, sn::Contracts) &&
               SlotTarget(base, sn::AudioSystemVtable, sn::TriggerIdSlot) == sn::GetAudioTriggerId &&
               SlotTarget(base, sn::EntityAudioProxyVtable, sn::ExecuteTriggerSlot) == sn::ProxyExecuteTrigger;
    } __except (EXCEPTION_EXECUTE_HANDLER) {
        return false;
    }
}
// Whether the live audio system is the CAudioSystem whose lookup the gate checked.
const char* AudioSystemKind(std::uintptr_t global, std::uintptr_t vtable)
{
    __try {
        const auto system = *reinterpret_cast<const std::uintptr_t*>(global);
        if (!system) { return "none"; }
        return *reinterpret_cast<const std::uintptr_t*>(system) == vtable ? "caudiosystem" : "other";
    } __except (EXCEPTION_EXECUTE_HANDLER) {
        return "fault";
    }
}
// Where the replay reads and what it compares, resolved once the gate holds. Tests
// put fakes here.
struct Natives {
    std::uintptr_t base = 0;
    std::uintptr_t entitySystem = 0, audioSystem = 0;   // the globals holding them
    std::uintptr_t audioSystemVtable = 0, proxyVtable = 0;
};
Natives natives{};
DWORD Resolve(Natives& n)
{
    n = {};
    const auto base = PreyBase();
    if (!base) { return ERROR_NOT_READY; }
    if (!GateBodies(base)) { return ERROR_BAD_EXE_FORMAT; }
    n.base = base;
    n.entitySystem = base + sn::EntitySystem;
    n.audioSystem = base + sn::AudioSystemGlobal;
    n.audioSystemVtable = base + sn::AudioSystemVtable;
    n.proxyVtable = base + sn::EntityAudioProxyVtable;
    return ERROR_SUCCESS;
}
bool Install(const Natives& n)
{
    void* originals[3]{};
    const hook::Spec hooks[] = {
        {reinterpret_cast<void*>(n.base + sn::PlaySoundEnter), reinterpret_cast<void*>(ClipEnterHook), &originals[0]},
        {reinterpret_cast<void*>(n.base + sn::TriggerIdByName), reinterpret_cast<void*>(TriggerIdHook), &originals[1]},
        {reinterpret_cast<void*>(n.base + sn::ProxyExecuteTrigger), reinterpret_cast<void*>(ExecuteHook), &originals[2]}};
    const auto result = hook::InstallAll(hooks, [&] {
        originalClipEnter = reinterpret_cast<ClipEnterFn>(originals[0]);
        originalTriggerId = reinterpret_cast<TriggerIdFn>(originals[1]);
        originalExecute = reinterpret_cast<ExecuteFn>(originals[2]);
    });
    if (!result) {
        Log(std::string("hooks result=") + hook::StepName(result.step) + " status=" + std::to_string(result.status) +
            " index=" + std::to_string(result.index));
        return false;
    }
    Log("hooks result=0 count=3");
    return true;
}
// Tests replace the engine here.
DWORD (*gResolve)(Natives&) = Resolve;
bool (*gInstall)(const Natives&) = Install;
std::atomic<bool> gated{false};
// True once the gate holds; refused for good on a mismatch.
bool Gate()
{
    if (gated.load()) { return true; }
    if (fault.load()) { return false; }
    Natives n{};
    const DWORD ready = gResolve(n);
    if (ready == ERROR_NOT_READY) { return false; }
    if (ready) {
        fault = true;
        gate = "refused";
        Log("gate result=" + std::to_string(ready) + " detail=bytes_mismatch");
        return false;
    }
    natives = n;
    audioSystem = AudioSystemKind(n.audioSystem, n.audioSystemVtable);
    gate = "ok";
    gated = true;
    Log("gate result=0 rows=" + std::to_string(std::size(sn::Contracts)) + " audioSystem=" + audioSystem.load());
    return true;
}

// --- the replay (main thread) -------------------------------------------------------
std::atomic<bool> playOn{true};
std::atomic<std::uint64_t> played{0}, playFailed{0};
std::atomic<const char*> lastPlay{"none"};
using GetEntityFn = std::uintptr_t(__fastcall*)(std::uintptr_t, std::uint32_t);
using GetProxyFn = std::uintptr_t(__fastcall*)(std::uintptr_t, int);
// SAudioCallBackInfos as the PlaySound clip passes it: nothing to notify, normal priority.
struct CallbackInfo { void* object; void* userData; void* owner; std::uint32_t flags; };
const CallbackInfo kNoCallback{nullptr, nullptr, nullptr, sn::CallbackFlags};
// The ids the audio system has confirmed: the trigger exists and its id is the
// captured one. Asked once each; a full list only means asking again.
std::array<std::uint32_t, 32> confirmed{};
bool Confirmed(std::uint32_t id) { return std::find(confirmed.begin(), confirmed.end(), id) != confirmed.end(); }
void Confirm(std::uint32_t id)
{
    if (const auto free = std::find(confirmed.begin(), confirmed.end(), 0u); free != confirmed.end()) { *free = id; }
}
// A set of triggers to play.
struct Sound { const char* what; std::span<const sn::Trigger> triggers; };
enum class Play : std::uint8_t { played, noEntity, noProxy, unknownTrigger, fault };
const char* PlayName(Play p)
{
    switch (p) {
    case Play::played: return "played";
    case Play::noEntity: return "no_entity";
    case Play::noProxy: return "no_proxy";
    case Play::unknownTrigger: return "unknown_trigger";
    case Play::fault: return "fault";
    }
    return "none";
}
template <class Fn> Fn Virtual(std::uintptr_t object, std::size_t slot)
{
    return reinterpret_cast<Fn>((*reinterpret_cast<const std::uintptr_t* const*>(object))[slot / sizeof(void*)]);
}
// The player's entity (the entity system's GetEntity), its audio proxy (GetProxy,
// which must be a CEntityAudioProxy: its ExecuteTrigger is the gated body), then
// each trigger on the proxy's default local id, as the clips played them.
Play PlayTriggers(const Natives& n, const Sound& s, unsigned& count)
{
    __try {
        const auto system = *reinterpret_cast<const std::uintptr_t*>(n.entitySystem);
        const auto entity = system ? Virtual<GetEntityFn>(system, sn::GetEntitySlot)(system, sn::PlayerEntityId) : 0;
        if (!entity) { return Play::noEntity; }
        const auto proxy = Virtual<GetProxyFn>(entity, sn::GetProxySlot)(entity, sn::AudioProxyType);
        if (!proxy || *reinterpret_cast<const std::uintptr_t*>(proxy) != n.proxyVtable) { return Play::noProxy; }
        const auto audio = *reinterpret_cast<const std::uintptr_t*>(n.audioSystem);
        for (std::size_t i = 0; i < s.triggers.size(); ++i) {
            const sn::Trigger& trigger = s.triggers[i];
            if (!Confirmed(trigger.id)) {
                std::uint32_t id = 0;
                if (!audio || *reinterpret_cast<const std::uintptr_t*>(audio) != n.audioSystemVtable ||
                    !Virtual<TriggerIdFn>(audio, sn::TriggerIdSlot)(audio, trigger.name, &id) || id != trigger.id) {
                    return Play::unknownTrigger;
                }
                Confirm(id);
            }
            Virtual<ExecuteFn>(proxy, sn::ExecuteTriggerSlot)(proxy, trigger.id, 0, sn::DefaultProxyId, &kNoCallback);
            ++count;
        }
        return Play::played;
    } __except (EXCEPTION_EXECUTE_HANDLER) {
        return Play::fault;
    }
}
bool PlayTriggerSet(const Sound& s)
{
    if (!playOn.load()) { return false; }
    unsigned count = 0;
    const Play result = Gate() ? PlayTriggers(natives, s, count) : Play::fault;
    lastPlay = PlayName(result);
    (result == Play::played ? played : playFailed).fetch_add(1);
    Log(std::string("play what=") + s.what + " result=" + PlayName(result) + " triggers=" + std::to_string(count) +
        (gated.load() ? std::string() : " gate=" + std::string(gate.load())));
    return result == Play::played;
}

void SetWeaponSoundCapture(unsigned on)
{
    wanted = on != 0;
    Log("settings capture=" + std::to_string(wanted.load()));
}
bool WeaponSoundCapturing() { return wanted.load(); }
void SetWeaponSoundPlay(unsigned on)
{
    playOn = on != 0;
    Log("settings play=" + std::to_string(playOn.load()));
}
unsigned WeaponSoundPlay() { return playOn.load() ? 1u : 0u; }
void WeaponSoundWindow(const char* what, std::uint32_t item, const char* className)
{
    if (!wanted.load()) { return; }
    const auto now = MonotonicNanoseconds();
    windowItem = item;
    windowLines = 0;
    windowEnd = now + kWindowNs;
    windowStart = now;
    ++windows;
    Log(std::string("window what=") + what + " class=" + className + " item=" + std::to_string(item) + " n=" +
        std::to_string(windows.load()) + " hooks=" + std::to_string(installed.load()));
}
unsigned WeaponSoundDrawFor(const char* className)
{
    unsigned draw = 0;
    for (const auto& c : sn::ClassDraws) {
        if (std::strcmp(c.className, className) == 0) { return draw; }
        ++draw;
    }
    return draw;   // past the table: the default
}
bool WeaponSoundPlayDraw(unsigned draw)
{
    return PlayTriggerSet({"draw", draw < std::size(sn::ClassDraws) ? sn::ClassDraws[draw].triggers
                                                                    : std::span<const sn::Trigger>(sn::DefaultDraw)});
}
bool WeaponSoundPlayStow() { return PlayTriggerSet({"stow", sn::StowTriggers}); }
void ServiceWeaponSoundFrame()
{
    if (wanted.load() && !installed.load() && !hookFailed.load() && Gate()) {
        if (gInstall(natives)) {
            installed = true;
        } else {
            hookFailed = true;
        }
    }
    const auto start = windowStart.load();
    if (start && MonotonicNanoseconds() > windowEnd.load()) {
        windowStart = 0;
        Log("window end item=" + std::to_string(windowItem.load()) + " lines=" +
            std::to_string(std::min(windowLines.load(), kMaxLines)) + " dropped=" + std::to_string(dropped.load()));
    }
}
std::string WeaponSoundReport()
{
    return " play=" + std::to_string(playOn.load()) + " played=" + std::to_string(played.load()) + " playFailed=" +
           std::to_string(playFailed.load()) + " lastPlay=" + lastPlay.load() + " capture=" + std::to_string(wanted.load()) +
           " hooks=" + std::to_string(installed.load()) + " hookFailed=" + std::to_string(hookFailed.load()) + " gate=" +
           gate.load() + " fault=" + std::to_string(fault.load()) + " audioSystem=" + audioSystem.load() + " windows=" +
           std::to_string(windows.load()) + " clips=" + std::to_string(clips.load()) + " resolves=" +
           std::to_string(resolves.load()) + " executes=" + std::to_string(executes.load()) + " dropped=" +
           std::to_string(dropped.load());
}
